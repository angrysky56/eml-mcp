Tests for the SQLite persistence layer (EMLFormulaDB).

Covers:
- Schema creation and seed initialization (DB-06)

- Derivation provenance tracking (DB-02)

from __future__ import annotations

import json
import math

import pytest

from eml_mcp.database import EMLFormulaDB
from eml_mcp.registry import SEED_FORMULAS, build_exp_tree, build_ln_tree
from eml_mcp.trees import EMLNode


TOLERANCE = 1e-10


@pytest.fixture
def db():
    """Create a fresh in-memory database for each test."""
    database = EMLFormulaDB(":memory:")
    yield database
    database.close()


class TestSchemaAndSeeding:
    """DB auto-creates schema and seeds all 8 formulas on init."""

    def test_seed_formula_count(self, db: EMLFormulaDB) -> None:
        """All 8 seed formulas must be present after initialization."""
        formulas = db.list_formulas()
        assert len(formulas) == 8

    def test_seed_formula_names(self, db: EMLFormulaDB) -> None:
        """All expected formula names must exist."""
        names = {f["name"] for f in db.list_formulas()}
        expected = set(SEED_FORMULAS.keys())
        assert names == expected

    def test_seed_formula_fields(self, db: EMLFormulaDB) -> None:
        """Each seeded formula has all required fields populated."""
        for formula in db.list_formulas():
            assert formula["description"]
            assert formula["tree_json"]
            assert formula["rpn"]
            assert formula["expression"]
            assert formula["depth"] >= 0
            assert formula["k"] >= 1
            assert formula["leaf_count"] >= 1

    def test_seed_derivation_records(self, db: EMLFormulaDB) -> None:
        """Each seed formula should have a derivation record with method='seed'."""
        for name in SEED_FORMULAS:
            derivations = db.get_derivations(name)
            assert len(derivations) >= 1
            assert derivations[0]["method"] == "seed"


class TestFormulaCRUD:
    """Test formula create/read operations."""

    def test_get_formula_exists(self, db: EMLFormulaDB) -> None:
        """get_formula returns data for existing formulas."""
        result = db.get_formula("exp")
        assert result is not None
        assert result["name"] == "exp"
        assert result["k"] == 3

    def test_get_formula_not_found(self, db: EMLFormulaDB) -> None:
        """get_formula returns None for non-existent formulas."""
        result = db.get_formula("nonexistent_formula")
        assert result is None

    def test_formula_exists(self, db: EMLFormulaDB) -> None:
        """formula_exists returns correct boolean."""
        assert db.formula_exists("exp") is True
        assert db.formula_exists("nonexistent") is False

    def test_add_formula(self, db: EMLFormulaDB) -> None:

        tree = build_exp_tree()  # reuse exp tree for test
        db.add_formula(
            name="test_formula",
            description="Test formula for unit tests",
            tree=tree,
            variables=["x"],
            note="test note",
        )
        result = db.get_formula("test_formula")
        assert result is not None
        assert result["name"] == "test_formula"
        assert result["description"] == "Test formula for unit tests"
        assert result["note"] == "test note"

    def test_add_duplicate_formula_raises(self, db: EMLFormulaDB) -> None:

        """get_verifications returns empty list for unknown formula."""

        assert records == []


class TestDerivationProvenance:

class TestEMLNodeRoundTrip:
    """EMLNode.from_dict must produce numerically identical trees."""

    @pytest.mark.parametrize("name", list(SEED_FORMULAS.keys()))
    def test_round_trip_structure(self, name: str) -> None:
        """to_dict -> from_dict preserves tree structure for all formulas."""
        entry = SEED_FORMULAS[name]
        original = entry["builder"]()
        reconstructed = EMLNode.from_dict(original.to_dict())

        assert reconstructed.depth == original.depth
        assert reconstructed.node_count == original.node_count
        assert reconstructed.leaf_count == original.leaf_count
        assert reconstructed.to_expression() == original.to_expression()
        assert reconstructed.to_rpn() == original.to_rpn()

    @pytest.mark.parametrize("name", ["exp", "ln", "e", "zero"])
    def test_round_trip_evaluation_univariate(self, name: str) -> None:
        """Deserialized tree evaluates identically for univariate formulas."""
        entry = SEED_FORMULAS[name]
        original = entry["builder"]()
        reconstructed = EMLNode.from_dict(original.to_dict())

        test_points = [complex(0.5772), complex(1.2824), complex(2.5)]
        variables = entry["variables"]

        for z in test_points:
            if variables:
                bindings = {variables[0]: z}
            else:
                bindings = {}
            orig_val = original.evaluate(bindings)
            recon_val = reconstructed.evaluate(bindings)
            diff = abs(orig_val - recon_val)
            assert diff < TOLERANCE, (
                f"{name} round-trip failed at {z}: "
                f"original={orig_val}, reconstructed={recon_val}, diff={diff}"
            )

    @pytest.mark.parametrize("name", ["subtract", "add", "multiply"])
    def test_round_trip_evaluation_bivariate(self, name: str) -> None:
        """Deserialized tree evaluates identically for bivariate formulas."""
        entry = SEED_FORMULAS[name]
        original = entry["builder"]()
        reconstructed = EMLNode.from_dict(original.to_dict())

        test_pairs = [(complex(0.5772), complex(1.2824)), (complex(1.5), complex(2.0))]
        for xv, yv in test_pairs:
            orig_val = original.evaluate({"x": xv, "y": yv})
            recon_val = reconstructed.evaluate({"x": xv, "y": yv})
            diff = abs(orig_val - recon_val)
            assert diff < TOLERANCE, (
                f"{name} round-trip failed at ({xv}, {yv}): "
                f"original={orig_val}, reconstructed={recon_val}, diff={diff}"
            )


class TestDBTreeDeserialization:
    """Test that trees loaded from DB evaluate correctly."""

    def test_db_formula_evaluates_correctly(self, db: EMLFormulaDB) -> None:
        """Formula loaded from DB evaluates identically to builder output."""
        row = db.get_formula("exp")
        assert row is not None

        db_tree = EMLNode.from_dict(json.loads(row["tree_json"]))
        builder_tree = build_exp_tree()

        test_val = complex(1.0)
        db_result = db_tree.evaluate({"x": test_val})
        builder_result = builder_tree.evaluate({"x": test_val})

        assert abs(db_result - builder_result) < TOLERANCE

    def test_db_ln_formula_evaluates_correctly(self, db: EMLFormulaDB) -> None:
        """ln formula loaded from DB matches math.log."""
        row = db.get_formula("ln")
        assert row is not None

        db_tree = EMLNode.from_dict(json.loads(row["tree_json"]))
        test_val = complex(2.5)
        result = db_tree.evaluate({"x": test_val}).real
        expected = math.log(2.5)

        assert abs(result - expected) < TOLERANCE
